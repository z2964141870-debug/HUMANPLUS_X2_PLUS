# Recommended clean-machine bootstrap (order matters):
#   conda create -n x2-sonic-isaaclab python=3.11.15 pip=26.1.2 -y
#   conda activate x2-sonic-isaaclab
#   python -m pip install "isaacsim[all,extscache]==5.1.0.0" --extra-index-url https://pypi.nvidia.com
#   git clone https://github.com/isaac-sim/IsaacLab.git
#   cd IsaacLab && git checkout 37ddf626871758333d6ed89cf64ad702aef127d0
#   ./isaaclab.sh --install
#   cd /path/to/CWI_CrossEmbodiment_Sim
#   python -m pip install -r requirements.txt
#   python -m pip install --no-deps "mink==1.2.0"
#
# Why Isaac Sim, IsaacLab and Mink are commands/comments instead of active pip
# lines below: the verified environment intentionally has upstream metadata
# conflicts. Isaac Sim 5.1 declares torch==2.7 while this project's working
# checkpoints/runtime use torch==2.8.0+cu128; Mink 1.2 declares mujoco>=3.8.1
# while the official X2 simulator is linked against mujoco==3.3.7. A single
# modern pip resolution therefore cannot reproduce the working stack. Installing
# the base first and applying the verified runtime pins afterward is deliberate.
# Never run an unconstrained `pip install -U` in this environment.
#
# NVIDIA wheels and Isaac Sim extensions are hosted outside the default index.

# NON-PIP REQUIREMENTS — install/copy these before running the project
# =============================================================================
#
# [A. Recommended host operating system]
#   - Ubuntu 22.04 LTS, x86_64.
#   - NVIDIA proprietary driver compatible with CUDA 12.8 (`nvidia-smi` must
#     work before installing Python packages).
#   - The validated workstation GPU is an RTX 5060 8 GB. More VRAM is useful;
#     training launchers must reduce NUM_ENVS if the target GPU is smaller.
#
# [B. Host system packages]
# Install with apt (package names are for Ubuntu 22.04):
#   git git-lfs curl wget jq rsync unzip zip tar
#   build-essential gcc g++ cmake ninja-build pkg-config patchelf
#   ffmpeg xvfb xauth
#   libgl1 libglib2.0-0 libx11-6 libxext6 libxrender1 libsm6
#   libxi6 libxrandr2 libxinerama1 libxcursor1 libvulkan1 vulkan-tools
#   mesa-utils
# Then run once:
#   git lfs install
#
# [C. Containers required for the official X2 closed-loop domain]
#   - Docker Engine.
#   - NVIDIA Container Toolkit and a working `docker run --gpus all ...`.
#   - The vendor/base image `g1-deploy-dev:latest`.
#   - The project image built by:
#       /home/humanplus/projects/ZHY/x2_official_rl_deploy_v1/docker/Dockerfile
#
# The vendor image is not obtainable from PyPI. Migrate it explicitly:
#   source host: docker save g1-deploy-dev:latest | gzip > g1-deploy-dev.tar.gz
#   target host: gunzip -c g1-deploy-dev.tar.gz | docker load
#
# The official derived container additionally installs:
#   apt: libgflags-dev python3-pip xvfb xauth
#        ros-humble-grid-map-msgs ros-humble-grid-map-ros
#        ros-humble-rmw-cyclonedds-cpp
#   pip inside that container only: numpy<2, onnxruntime==1.16.3
# Do NOT replace that container's onnxruntime with the host GPU build above.
#
# [D. ROS / AimDK]
#   - ROS 2 Humble is used inside the official X2 container.
#   - `aimdk_msgs`, `nav_msgs`, `sensor_msgs`, `rclpy` and the X2 controller
#     executable come from the official AimDK/ROS workspace, not from PyPI.
#   - Keep the official CycloneDDS middleware configuration.
#   - Do not mix host ROS Jazzy Python paths into this Python 3.11 environment.
#   - Use ROS_DOMAIN_ID <= 232; domains 233/234 overflow the DDS UDP port range
#     in this setup and were observed to fail before physics startup.
#
# [E. Required source trees / local Python modules]
# These modules are imported by the project but are not public PyPI packages.
# Preserve them at stable paths, or add their roots to PYTHONPATH:
#   1. This repository:
#      /home/humanplus/projects/ZHY/CWI_CrossEmbodiment_Sim
#   2. X2 official deployment SDK v1.0:
#      /home/humanplus/projects/ZHY/x2_official_rl_deploy_v1
#   3. Legacy X2/SONIC implementation and assets:
#      /home/humanplus/x2_teleop_final/x2_sonic
#      (provides x2_sonic and sonic_x2_sandbox modules/assets)
#   4. Human+ motion-capture workspace:
#      /home/humanplus/projects/Human+智能服装动作捕捉系统
#      (provides the audited general_motion_retargeting package and X2 config)
#   5. GR00T Whole-Body Control / SONIC workspace when its loaders or deploy
#      adapters are used:
#      /home/humanplus/sonic_lty/GR00T-WholeBodyControl
#      (provides gear_sonic and related modules/assets)
#
# Suggested portable layout on the new machine:
#   $X2_WORKSPACE/CWI_CrossEmbodiment_Sim
#   $X2_WORKSPACE/x2_official_rl_deploy_v1
#   $X2_WORKSPACE/x2_sonic
#   $X2_WORKSPACE/Human+智能服装动作捕捉系统
